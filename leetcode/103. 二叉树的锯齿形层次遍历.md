整体思路是层序遍历，但是要控制加入的顺序

这里我的思路是找到一个矛盾点，上一层先访问的节点，他的孩子节点实际要后添加，而且节点添加的顺序和整体的顺序还有矛盾，需要先右再左，还要插入头部。然后下一层又是先左再右，进行尾插，控制复杂，所以为了解决这个问题，插入顺序不变，控制访问顺序更优雅

			else:
				node = queue.popright()
			level_res.append(node.val)
			if node.left:
				queue.append(node.left)
			if node.right:
				queue.append(node.right)
			level += 1
		res.append(level_res)
	return res

import collections

class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []
            
        res = []
        queue = collections.deque([root])
        is_left_to_right = True  # 反转开关，初始第一层是从左到右
        
        while queue:
            level_res = []
            # 这里的遍历永远是铁打不变的从左到右
            for _ in range(len(queue)):
                node = queue.popleft()
                level_res.append(node.val)
                
                if node.left: queue.append(node.left)
                if node.right: queue.append(node.right)
            
            # 核心魔法：根据开关决定是否反转当前层
            if not is_left_to_right:
                level_res.reverse()  # Python 底层 C 实现，极速反转
                
            res.append(level_res)
            
            # 这一层搞定了，开关拨到另一边
            is_left_to_right = not is_left_to_right
            
        return res
