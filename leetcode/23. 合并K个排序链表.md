class ListNode:
	def __init__(self, val=0, next=None):
		self.val = val
		self.next = next
		
def mergeKList(lists: List[Optional[ListNode]]) -> Optional[ListNode]:
	def helper(lists: List[Optional[ListNode]], start: int, end: int) -> Optional[ListNode]:
		if start > end:
			return None
			
		if start == end:
			return lists[start]

		left = helper(lists, start, mid)

def mergeList(l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
	dummy = ListNode(0)
	pre = dummy
	while l1 or l2:
		if not l1:
			pre.next = l2
			break
		if not l2:
			pre.next = l1
			break
		if l1.val < l2.val:
			pre.next = l1
			pre = l1
			l1 = l1.next
		else:
			pre.next = l2
			pre = l2
			l2 = l2.next
	return dummy.next
