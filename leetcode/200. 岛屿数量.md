
从第一个元素开始遍历，如果遇到陆地，岛屿数量+1，进行四个方向的dfs扩展，每个方向直到遇到边界或者遇到水的时候停止，如果遇到陆地就用水把它淹没


```
from typing import List

def numIslands(grid: List[List[str]]) -> int:
    if not grid:
        return 0

    rows, cols = len(grid), len(grid[0])
    count = 0

    def dfs(r: int, c: int) -> None:
        # 越界或海水/访问过，停止
        if (
            r < 0 or r >= rows
            or c < 0 or c >= cols
            or grid[r][c] != "1"
        ):
            return

        grid[r][c] = "0"  # 标记已访问

        dfs(r + 1, c)
        dfs(r - 1, c)
        dfs(r, c + 1)
        dfs(r, c - 1)

    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == "1":
                count += 1
                dfs(r, c)  # 淹没这一整座岛

    return count
```
